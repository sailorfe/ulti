** installation

since this port is a subdirectory, you'll have to point neovim here so it can find =lua/ulti.lua=. without a package manager:

#+BEGIN_SRC sh
  # as submodule in version-controlled dotfiles
  git submodule add https://codeberg.org/sailorfe/ulti pack/plugins/start
  # or 
  git clone https://codeberg.org/sailorfe/ulti .local/share/nvim/site/pack/plugins/start
#+END_SRC

then add =extras/nvim= to your =packpath=.

#+BEGIN_SRC lua
local ulti_path = vim.fn.stdpath("data") .. "/site/pack/plugins/start/ulti/extras/nvim"

vim.opt.rtp:append(ulti_path)
package.path = package.path .. ";" .. ulti_path .. "/lua/?.lua;" .. ulti_path .. "/lua/?/init.lua"

require("ulti").setup()
#+END_SRC

*** lazy.nvim

#+BEGIN_SRC lua
{  
  "https://codeberg.org/sailorfe/ulti",
  name = "ulti",  
  config = function(plugin)  
    vim.opt.rtp:append(plugin.dir .. "/extras/nvim")  
  end,  
}
#+END_SRC

*** packer.nvim

#+BEGIN_SRC lua
{ use 'https://codeberg.org/sailorfe/ulti', rtp = 'extras/nvim' }
#+END_SRC
